
Let us now consider

 \[\begin{align*}&\left|\int_{0}^{T}\langle\mathbf{B}_{n}(\boldsymbol{u}_{n}(t)),\mathbf{P}_{n}\boldsymbol{v}(t)\rangle\mathrm{d}t-\int_{0}^{T}\langle\mathbf{B}(\boldsymbol{u}(t)),\boldsymbol{v}(t)\rangle\mathrm{d}t\right|\\&\leq\left|\int_{0}^{T}\langle\mathbf{B}(\boldsymbol{u}_{n}(t)),\mathbf{P}_{n}\boldsymbol{v}(t)-\boldsymbol{v}(t)\rangle\mathrm{d}t\right|+\left|\int_{0}^{T}\langle\mathbf{B}(\boldsymbol{u}_{n}(t))-\mathbf{B}(\boldsymbol{u}(t)),\boldsymbol{v}(t)-\boldsymbol{\psi}_{\varepsilon}(t)\rangle\mathrm{d}t\right|\\&\quad+\left|\int_{0}^{T}\langle\mathbf{B}(\boldsymbol{u}_{n}(t))-\mathbf{B}(\boldsymbol{u}(t)),\boldsymbol{\psi}_{\varepsilon}(t)\rangle\mathrm{d}t\right|\\&\leq\|\mathbf{P}_{n}-I_{\mathbb{d}}\|_{\mathcal{L}(\mathbb{V})}\left(\int_{0}^{T}\|\boldsymbol{u}_{n}(t)\|_{\mathbb{L}^{4}}^{\frac{8}{\mathbb{L}^{4}}}\mathrm{d}t\right)^{\frac{d}{4}}\left(\int_{0}^{T}\|\boldsymbol{v}(t)\|_{\mathbb{V}}^{\frac{4}{4-d}}\mathrm{d}t\right)^{\frac{4-d}{4}}\\&\quad+C\left[\int_{0}^{T}\left(\|\boldsymbol{u}_{n}(t)\|_{\mathbb{L}^{4}}^{\frac{8}{\mathbb{L}^{4}}}+\|\boldsymbol{u}(t)\|_{\mathbb{L}^{4}}^{\frac{8}{\mathbb{L}^{4}}}\right)\mathrm{d}t\right]^{\frac{d}{4}}\left(\int_{0}^{T}\|\boldsymbol{v}(t)-\boldsymbol{\psi}_{\varepsilon}(t)\|_{\mathbb{V}}^{\frac{4}{4-d}}\mathrm{d}t\right)^{\frac{4-d}{4}}\\&\quad+\left|\int_{0}^{T}\langle\mathbf{B}(\boldsymbol{u}_{n}(t))-\mathbf{B}(\boldsymbol{u}(t)),\boldsymbol{\psi}_{\varepsilon}(t)\rangle\mathrm{d}t\right|\\&\leq C\Bigg[\int_{0}^{T}\left(\|\boldsymbol{u}_{n}(t)\|_{\mathbb{L}^{4}}^{\frac{8}{\mathbb{L}^{4}}}+\|\boldsymbol{u}(t)\|_{\mathbb{L}^{4}}^{\frac{8}{\mathbb{L}^{4}}}\right)\mathrm{d}t\Bigg]^{\frac{d}{4}}\left(\int_{0}^{T}\|\boldsymbol{v}(t)-\boldsymbol{\psi}_{\varepsilon}(t)\|_{\mathbb{V}}^{\frac{4}{4-d}}\mathrm{d}t\right)^{\frac{4-d}{4}}\quad\mathrm{as}\ n\to\infty,\end{align*}\] 

Since an application of Ladyzheskaya’s inequality and the energy estimate (3.6) yield

 \[\int_{0}^{T}\|\boldsymbol{u}_{n}(t)\|_{\mathbb{L}^{4}}^{\frac{8}{\mathbb{L}^{4}}}\mathrm{d}t\leq C\sup_{t\in[0,T]}\|\boldsymbol{u}_{n}(t)\|_{\mathbb{H}}^{\frac{2(4-d)}{d}}\int_{0}^{T}\|\boldsymbol{u}_{n}(t)\|_{\mathbb{V}}^{2}\mathrm{d}t\leq\left\{\|\boldsymbol{u}_{0}\|_{\mathbb{H}}^{2}+\frac{1}{\mu}\int_{0}^{T}\|\boldsymbol{f}(t)\|_{\mathbb{V}^{\prime}}^{2}\mathrm{d}t\right\}^{\frac{4}{d}},\] 

so that

 \[\begin{align*}&\left|\int_{0}^{T}\langle\mathbf{B}_{n}(\boldsymbol{u}_{n}(t)),\mathbf{P}_{n}\boldsymbol{v}(t)\rangle\mathrm{d}t-\int_{0}^{T}\langle\mathbf{B}(\boldsymbol{u}(t)),\boldsymbol{v}(t)\rangle\mathrm{d}t\right|\\&\leq C\Bigg\{\|\boldsymbol{u}_{0}\|_{\mathbb{H}}^{2}+\frac{1}{\mu}\int_{0}^{T}\|\boldsymbol{f}(t)\|_{\mathbb{V}^{\prime}}^{2}\mathrm{d}t\Bigg\}\varepsilon.\end{align*}\] 

Since  \( \varepsilon > 0 \)  is arbitrary, we deduce that

 \[\lim_{n\to\infty}\int_{0}^{T}\langle\mathbf{B}_{n}(\boldsymbol{u}_{n}(t)),\mathbf{P}_{n}\boldsymbol{v}(t)\rangle\mathrm{d}t=\int_{0}^{T}\langle\mathbf{B}(\boldsymbol{u}(t)),\boldsymbol{v}(t)\rangle\mathrm{d}t.\] 

We know from (3.6) that

 \[\int_{0}^{T}\|\mathcal{C}_{n}(\boldsymbol{u}_{n}(t)\|_{\widetilde{\mathbb{L}^{\frac{r+1}{r}}}}^{\frac{r+1}{r-1}}d t\leq\int_{0}^{T}\|\boldsymbol{u}_{n}(t)\|_{\widetilde{\mathbb{L}^{r+1}}}^{r+1}d t\leq\|\boldsymbol{u}_{0}\|_{\mathbb{H}}^{2}+\frac{1}{\mu}\int_{0}^{T}\|\boldsymbol{f}(t)\|_{\mathbb{V}^{\prime}}^{2}d t.\] 

Therefore, by an application of the Banach-Alaoglu theorem yields the existence of a  \( \xi \in L^{\frac{r+1}{r}}(0, T; \widetilde{\mathbb{L}^{\frac{r+1}{r}}}) \)  such that

 \[\mathcal{C}_{n}(\boldsymbol{u}_{n})\xrightarrow{w}\boldsymbol{\xi}\mathrm{~i n~}\mathrm{~L}^{\frac{r+1}{r}}(0,T;\widetilde{\mathbb{L}^{\frac{r+1}{r}}}).\]
