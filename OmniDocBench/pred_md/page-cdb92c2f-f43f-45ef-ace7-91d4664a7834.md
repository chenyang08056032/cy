
Let us consider that  \( (\mathcal{L},[\cdot_{\lambda}],\alpha) \)  is a Hom-Leibniz conformal algebra, which has coefficients in the representation  \( (l,r,\mathcal{M},\beta) \) . The group of nth-cochain  \( C_{HomL}^{n}(\mathcal{L},\mathcal{M}) \)  contains the map  \( f_{\lambda_{1},\cdots,\lambda_{n-1}}:\wedge^{\otimes n}\mathcal{L}\to\mathbb{C}[\lambda_{1},\cdots,\lambda_{n-1}]\otimes\mathcal{M} \)  defined by  \( p_{1}\otimes p_{2}\otimes\cdots\otimes p_{n}\mapsto f_{\lambda_{1},\lambda_{2},\cdots,\lambda_{n-1}}(p_{1},p_{2},\cdots,p_{n}) \) . These maps satisfy the following conformal sesqui-linearity condition:

 \[\begin{align*}f_{\lambda_{1},\lambda_{2},\cdots,\lambda_{n-1}}(p_{1},p_{2},\cdots,\partial(p_{i}),\cdots,p_{n})\\=\begin{cases}-\lambda_{i}f_{\lambda_{1},\lambda_{2},\cdots,\lambda_{n-1}}(p_{1},p_{2},\cdots,p_{n}),&i=1,\cdots,n-1,\ $ \partial+\lambda_{1}+\lambda_{2}+\cdots+\lambda_{n-1})f_{\lambda_{1},\lambda_{2},\cdots,\lambda_{n-1}}(p_{1},p_{2},\cdots,p_{n}),&i=n.\end{cases}\end{align*}\] 

such that  \( f \circ \alpha^{\otimes n} = \beta \circ f \) . The coboundary map  \( \delta_{Hom} : C_{Hom}^{n}(\mathcal{L}, \mathcal{M}) \to C_{Hom}^{n}(\mathcal{L}, \mathcal{M}) \)  can be defined by

 \[\begin{align*}&(\delta_{HomL}f)_{\lambda_{1},\cdots,\lambda_{n}}(p_{1},\cdots,p_{n+1})\\=&\sum_{i=1}^{n}(-1)^{i+1}l(\alpha^{n-1}(p_{i}))_{\lambda_{i}}f_{\lambda_{1},\cdots,\dot{\lambda_{i}},\cdots,\lambda_{n}}(p_{1},\cdots,\hat{p_{i}},\cdots,p_{n+1})\\&+(-1)^{n+1}r(f_{\lambda_{1},\cdots,\lambda_{n-1}}(p_{1},\cdots,p_{n}))_{\lambda_{1}+\cdots+\lambda_{n}}\alpha^{n-1}(p_{n+1})\\&+\sum_{i<j}(-1)^{i+j}f_{\lambda_{i}+\lambda_{j},\lambda_{1},\cdots,\dot{\lambda_{i}},\cdots,\dot{\lambda_{j}},\cdots,\lambda_{n}}([p_{i}\lambda_{i}p_{j}],\alpha(p_{1}),\cdots,\alpha(p_{i}),\cdots,\alpha(p_{j}),\cdots,\alpha(p_{n+1})),\end{align*}\] 

where  \(  f \in C_{HomL}^{n}(\mathcal{L}, \mathcal{M})  \)  and  \( p_{1}, p_{2}, \cdots, p_{n}, p_{n+1} \in L \) . With coefficients in the representation  \(  (l, r, \mathcal{M}, \beta)  \) , the corresponding cohomology groups are known as the cohomology of the Hom-Leibniz conformal algebra  \(  (\mathcal{L}, [\cdot_{\lambda}], \alpha)  \) . If we assume that  \(  l(p)_{\lambda} m = [p_{\lambda} m]  \)  and  \(  r(m)_{\lambda} p = [m_{\lambda} p]  \)  then above equations turn to be

 \[\begin{align*}&(\delta_{HomL}f)_{\lambda_{1},\cdots,\lambda_{n}}(p_{1},\cdots,p_{n+1})\\=&\sum_{i=1}^{n}(-1)^{i+1}[\alpha^{n-1}(p_{i})_{\lambda_{i}}f_{\lambda_{1},\cdots,\dot{\lambda_{i}},\cdots,\lambda_{n}}(p_{1},\cdots,\hat{p_{i}},\cdots,p_{n+1})]\\&+(-1)^{n+1}[f_{\lambda_{1},\cdots,\lambda_{n-1}}(p_{1},\cdots,p_{n})_{\lambda_{1}+\cdots+\lambda_{n}}\alpha^{n-1}(p_{n+1})]\\&+\sum_{i<j}(-1)^{i+j}f_{\lambda_{i}+\lambda_{j},\lambda_{1},\cdots,\dot{\lambda_{i}},\cdots,\dot{\lambda_{j}},\cdots,\lambda_{n}}([p_{i}\lambda_{i}p_{j}],\alpha(p_{1}),\cdots,\alpha(\hat{p_{i}}),\cdots,\alpha(\hat{p_{j}}),\cdots,\alpha(p_{n+1})),\end{align*}\] 

Theorem 3.1. \(\delta_{HomL}^{2} \circ \delta_{HomL}^{1} = 0\).

Proof. Consider that for n = 1, we have

 \[\begin{align*}(\delta_{HomL}^{1}f)(p_{1},p_{2})&=l(p_{1})_{\lambda}f(p_{2})+r(f(p_{1}))_{\lambda}p_{2}-f([p_{1}\lambda p_{2}])\\=&[p_{1}\lambda f(p_{2})]+[f(p_{1})\lambda p_{2}]-f([p_{1}\lambda p_{2}]).\end{align*}\] 

and

 \[\begin{align*}(\delta_{HomL}^{2}f)_{\lambda_{1},\lambda_{2}}(p_{1},p_{2},p_{3})=&l(\alpha(p_{1}))_{\lambda_{1}}f_{\lambda_{2}}(p_{2},p_{3})-l(\alpha(p_{2}))_{\lambda_{2}}f_{\lambda_{1}}(p_{1},p_{3})\\&-r(f_{\lambda_{1}}(p_{1},p_{2}))_{\lambda_{1}+\lambda_{2}}\alpha(p_{3})-f_{\lambda_{1}+\lambda_{2}}([p_{1}\lambda_{1}p_{2}],\alpha(p_{3}))\\&+f_{\lambda_{1}+\lambda_{2}}([p_{1}\lambda_{1},p_{3}],\alpha(p_{2}))-f_{\lambda_{1}+\lambda_{2}}([p_{2}\lambda_{2}p_{3}],\alpha(p_{1}))\\=&[(\alpha(p_{1}))_{\lambda_{1}}f_{\lambda_{2}}(p_{2},p_{3})]-[(\alpha(p_{2}))_{\lambda_{2}}\delta_{HomL}^{1}f_{\lambda_{1}}(p_{1},p_{3})]\\&-[(\delta_{HomL}^{1}f_{\lambda_{1}}(p_{1},p_{2}))_{\lambda_{1}+\lambda_{2}}\alpha(p_{3})]-\delta_{HomL}^{1}f_{\lambda_{1}+\lambda_{2}}([p_{1}\lambda_{1}p_{2}],\alpha(p_{3}))\\&+\delta_{HomL}^{1}f_{\lambda_{1}+\lambda_{2}}([p_{1}\lambda_{1}p_{3}],\alpha(p_{2}))-\delta_{HomL}^{1}f_{\lambda_{1}+\lambda_{2}}([p_{2}\lambda_{2}p_{3}],\alpha(p_{1})).\end{align*}\] 

So we have

 \[\begin{aligned}&(\delta_{HomL}^{2}\circ\delta_{HomL}^{1}f)_{\lambda_{1},\lambda_{2}}(p_{1},p_{2},p_{3})\\&=[(\alpha(p_{1}))_{\lambda_{1}}\delta_{HomL}^{1}f_{\lambda_{2}}(p_{2},p_{3})]-[(\alpha(p_{2}))_{\lambda_{2}}\delta_{HomL}^{1}f_{\lambda_{1}}(p_{1},p_{3})]\\&-[(\delta_{HomL}^{1}f_{\lambda_{1}}(p_{1},p_{2}))_{\lambda_{1}+\lambda_{2}}\alpha(p_{3})]-\delta_{HomL}^{1}f_{\lambda_{1}+\lambda_{2}}([p_{1}\lambda_{1}p_{2}],\alpha(p_{3}))\\&+\delta_{HomL}^{1}f_{\lambda_{1}+\lambda_{2}}([p_{1}\lambda_{1}p_{3}],\alpha(p_{2}))-\delta_{HomL}^{1}f_{\lambda_{1}+\lambda_{2}}([p_{2}\lambda_{2}p_{3}],\alpha(p_{1}))\\=&[(\alpha(p_{1}))_{\lambda_{1}}\{[p_{2}\lambda_{2}f(p_{3})]+[f(p_{2})_{\lambda_{2}}p_{3}]-f([p_{2}\lambda_{2}p_{3})]\}\\&-[(\alpha(p_{2}))_{\lambda_{2}}\{[p_{1}\lambda_{1}f(p_{3})]+[f(p_{1})_{\lambda_{1}}p_{3}]-f([p_{1}\lambda_{1}p_{3}])]\}\\&-[(\{[p_{1}\lambda_{1}f(p_{2})]+[f(p_{1})_{\lambda_{1}}p_{2}]-f([p_{1}\lambda_{1}p_{2}])\}])_{\lambda_{1}+\lambda_{2}}\alpha(p_{3})]\\&-\{[[p_{1}\lambda_{1}p_{2}]_{\lambda_{1}+\lambda_{2}}f(\alpha(p_{3}))]+[f([p_{1}\lambda_{1}p_{2}])_{\lambda_{1}+\lambda_{2}}\alpha(p_{3})]-f([[p_{1}\lambda_{1}p_{2}]_{\lambda_{1}+\lambda_{2}}\alpha(p_{3})])]\}\\&+{[[p_{1}\lambda_{1}p_{3}]_{\lambda_{1}+\lambda_{2}}f(\alpha(p_{2}))]}+{[f([p_{1}\lambda_{1}p_{3}])_{\lambda_{1}+\lambda_{2}}\alpha(p_{2})]-f([[p_{1}\lambda_{1}p_{3}]_{\lambda_{1}+\lambda_{2}}\alpha(p_{2})])]\}\end{aligned}\]
