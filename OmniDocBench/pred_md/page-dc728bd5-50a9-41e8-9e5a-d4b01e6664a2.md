
We claim that

 \[\int_{Q}|g\lambda|\prod_{i\in\tau}\Big(\sum_{P\in\tilde{S},P\subseteq Q}\eta_{i,P}\chi_{P}(x)\Big)^{k_{i}-t_{i}}\lesssim\int_{Q}A_{\tilde{S},\eta_{\tau(1)}}^{k_{\tau(1)}-t_{\tau(1)}}\big(A_{\tilde{S},\eta_{\tau(2)}}^{k_{\tau(2)}-t_{\tau(2)}}\big(A_{\tilde{S},\eta_{\tau(3)}}^{k_{\tau(3)}-t_{\tau(3)}}\big|g\lambda|\big)\Big).\] 

Clearly, this follows directly from the definitions that

 \[\begin{aligned}&\int_{Q}|g\lambda|\prod_{i\in\tau}\Big(\sum_{P\in\tilde{S},P\subseteq Q}\eta_{i,P}\chi_{P}(x)\Big)^{k_{i}-t_{i}}\\ \lesssim&\int_{Q}A_{\tilde{S},\eta_{\tau(1)}}^{k_{\tau(1)}-t_{\tau(1)}}\Big(|g\lambda|\Big(\sum_{P\in\tilde{S},P\subseteq Q}\eta_{\tau(2),P}\chi_{P}(x)\Big)^{k_{\tau(2)}-t_{\tau(2)}}\Big(\sum_{P\in\tilde{S},P\subseteq Q}\eta_{\tau(3),P}\chi_{P}(x)\Big)^{k_{\tau(3)}-t_{\tau(3)}}\Big)\\ =&\int_{Q}A_{\tilde{S},\eta_{\tau(1)}}^{k_{\tau(1)}-t_{\tau(1)}-1}\Big(A_{\tilde{S}}\big(|g\lambda|\Big(\sum_{P\in\tilde{S},P\subseteq Q}\eta_{\tau(2),P}\chi_{P}\Big)^{k_{\tau(2)}-t_{\tau(2)}}\Big(\sum_{P\in\tilde{S},P\subseteq Q}\eta_{\tau(3),P}\chi_{P}\Big)^{k_{\tau(3)}-t_{\tau(3)}}\Big)\eta_{\tau(1)}\Big)\\ =&\int_{Q}A_{\tilde{S},\eta_{\tau(1)}}^{k_{\tau(1)}-t_{\tau(1)}-1}\Big(\eta_{\tau(1)}\sum_{Q^{\prime}\in\tilde{S}}\Big\langle|g\lambda|\Big(\sum_{P\in\tilde{S},P\subseteq Q}\eta_{\tau(2),P}\chi_{P}\Big)^{k_{\tau(2)}-t_{\tau(2)}}\Big(\sum_{P\in\tilde{S},P\subseteq Q}\eta_{\tau(3),P}\chi_{P}\Big)^{k_{\tau(3)}-t_{\tau(3)}}\Big\rangle_{Q^{\prime}}\chi_{Q^{\prime}}\Big)^{k_{\tau(3)}-t_{\tau(3)}}\Bigg)\\ \overset{(4.22)}{\lesssim}&\int_{Q}A_{\tilde{S},\eta_{\tau(1)}}^{k_{\tau(1)}-t_{\tau(1)}-1}\Big(\Big(\sum_{Q^{\prime}\in\tilde{S}}\Big\langle A_{\tilde{S},\eta_{\tau(2)}}^{k_{\tau(2)}-t_{\tau(2)}}\big(A_{\tilde{S},\eta_{\tau(3)}}^{k_{\tau(3)}-t_{\tau(3)}}\big|g\lambda|\big)\Big\rangle_{Q^{\prime}}\chi_{Q^{\prime}}\Big)\eta_{\tau(1)}\Bigg)\\ =&\int_{Q}A_{\tilde{S},\eta_{\tau(1)}}^{k_{\tau(1)}-t_{\tau(1)}}\big(A_{\tilde{S},\eta_{\tau(2)}}^{k_{\tau(2)}-t_{\tau(2)}}\big(A_{\tilde{S},\eta_{\tau(3)}}^{k_{\tau(3)}-t_{\tau(3)}}\big|g\lambda|\big)\Big).\end{aligned}\] 

Combining above estimate (4.27), we obtain

 \[\begin{align*}&\sum_{Q\in\mathcal{S}}\mu(Q)^{\eta}\left(\int_{Q}\big|g\lambda\big|\prod_{i\in\tau}|b_{i}-b_{i,Q}|^{k_{i}-t_{i}}\right)\prod_{i\in\tau}\langle|f_{i}(b_{i}-b_{i,Q})^{t_{i}}|\rangle_{Q}\prod_{j\in\tau}\langle|f_{j}|\rangle_{Q}\\\lesssim&\prod_{i\in\tau}\|b_{i}\|^{k_{i}}_{\mathrm{BMO}_{n_{i}}}\int_{Q}A_{\tilde{S},\eta_{\tau(1)}}^{k_{\tau(1)}-t_{\tau(1)}}\big(A_{\tilde{S},\eta_{\tau(2)}}^{k_{\tau(2)}-t_{\tau(2)}}\big(A_{\tilde{S},\eta_{\tau(3)}}^{k_{\tau(3)}-t_{\tau(3)}}|g\lambda|\big)\big)A_{\eta,\tilde{S}}(\vec{\mathbf{A}})\end{align*}\] 

Since  \( A_{\tilde{S}} \)  is self-adjoint, we conclude

 \[\begin{align*}&=\prod_{i\in\tau}\|b_{i}\|_{\mathrm{BMO}_{n_{i}}}^{k_{i}}\int_{X}A_{\tilde{S}}\left(A_{\tilde{S},\eta_{\tau(1)}}^{k_{\tau(1)}-t_{\tau(1)}-1}\left(A_{\eta,\tilde{S},\eta_{\tau(1)}}(\vec{\mathbf{A}})\right)\right)A_{\tilde{S},\eta_{\tau(2)}}^{k_{\tau(2)}-t_{\tau(2)}}\left(A_{\tilde{S},\eta_{\tau(3)}}^{k_{\tau(3)}-t_{\tau(3)}}|g\lambda|\right)\\&=\prod_{i\in\tau}\|b_{i}\|_{\mathrm{BMO}_{n_{i}}}^{k_{i}}\int_{X}A_{\tilde{S}}\left(A_{\tilde{S},\eta_{\tau(3)}}^{k_{\tau(3)}-t_{\tau(3)}}\left(A_{\tilde{S},\eta_{\tau(2)}}^{k_{\tau(2)}-t_{\tau(2)}}\left(A_{\tilde{S},\eta_{\tau(1)}}^{k_{\tau(1)}-t_{\tau(1)}-1}\left(A_{\eta,\tilde{S},\eta_{\tau(1)}}(\vec{\mathbf{A}})\right)\right)\right)|g\lambda|,\end{align*}\] 

where  \( \mathcal{A}_{\eta,\tilde{S},\eta_{\tau(1)}}(\vec{\mathbf{A}}) \)  means that  \( \mathcal{A}_{\eta,\tilde{S}}(\vec{\mathbf{A}})\eta_{\tau(1)} \)  and

 \[\vec{\mathbf{A}}=\Big(A_{\tilde{S},\eta_{\tau(1)}}^{t_{\tau(1)}}\big(|f_{\tau(1)}|),A_{\tilde{S},\eta_{\tau(2)}}^{t_{\tau(2)}}\big(|f_{\tau(2)}|),A_{\tilde{S},\eta_{\tau(3)}}^{t_{\tau(3)}}\big(|f_{\tau(3)}|),f_{\tau(4)}\Big).\]
