
Definition 3.1. Let  \( (\mathfrak{M}_{\theta},\rho) \)  be a representation of an averaging Lie conformal algebra  \( (\mathfrak{L},[\cdot_{\lambda}\cdot]_{P}) \) . Define a map  \( \xi^{p}:C^{p}(\mathfrak{L},\mathfrak{M})\to C_{AO}^{p}(\mathfrak{L},\mathfrak{M}) \)  by

 \[\xi^{p}f_{\lambda_{1},\lambda_{2},\cdots,\lambda_{p-1}}(x_{1},x_{2},\cdots,x_{p})=f_{\lambda_{1},\lambda_{2},\cdots,\lambda_{p-1}}(P x_{1},P x_{2},\cdots,P x_{p})-(\phi\circ f)_{\lambda_{1},\lambda_{2},\cdots,\lambda_{p-1}}(P x_{1},x_{2},\cdots,x_{p}),\] 

for all  \( x_{1},\cdots,x_{p}\in L \)  and  \( f\in C^{p}(\mathfrak{L},\mathfrak{M}) \) .

Lemma 3.2. For any  \( x_{1},\cdots,x_{p+1}\in\mathfrak{L} \)  and  \( f\in C^{p}(\mathfrak{L},\mathfrak{M}) \) , we have

 \[\xi^{p+1}(\delta^{p}(f))_{\lambda_{1},\lambda_{2},\cdots,\lambda_{p}}(x_{1},x_{2},\cdots,x_{p+1})=\partial^{p}(\xi^{p}(f))_{\lambda_{1},\lambda_{2},\cdots,\lambda_{p}}(x_{1},x_{2},\cdots,x_{p+1}).\] 

Proof. First, consider that

 \[\begin{aligned}\xi^{p+1}(\delta^{p}(f))_{\lambda_{1},\lambda_{2},\cdots,\lambda_{p}}&(x_{1},x_{2},x_{3},\cdots,x_{p+1})\\&=(\delta^{p}f)_{\lambda_{1},\lambda_{2},\cdots,\lambda_{p}}(Px_{1},Px_{2},\cdots,Px_{p},Px_{p+1})\\&\quad-(\phi\circ(\delta^{p}f))_{\lambda_{1},\lambda_{2},\cdots,\lambda_{p}}(Px_{1},x_{2},\cdots,x_{p},x_{p+1})\\&=\sum_{i=1}^{p+1}(-1)^{i+1}[(Px_{i})_{\lambda_{i}}f_{\lambda_{2},\cdots,\lambda_{i},\cdots,\lambda_{p}}(Px_{1},\cdots P\acute{x}_{i},\cdots,Px_{p+1})]\\&\quad+\sum_{1\leq i<j\leq p+1}(-1)^{i+j}f_{\lambda_{i}+\lambda_{j},\lambda_{1},\cdots,\lambda_{i},\cdots,\lambda_{j},\cdots,\lambda_{p}}([Px_{i\lambda_{i}}Px_{j}],Px_{1},\cdots,P\acute{x}_{i},\cdots,P\acute{x}_{j},\cdots,Px_{p+1})\\&\quad-\phi([(Px_{1})_{\lambda_{1}}f_{\lambda_{2},\cdots,\lambda_{i},\cdots,\lambda_{p}}(x_{2},\cdots,x_{p+1})]\\&\quad+\sum_{1<j\leq p+1}(-1)^{1+j}f_{\lambda_{1}+\lambda_{j},\lambda_{2},\cdots,\lambda_{j},\cdots,\lambda_{p}}([Px_{1\lambda_{1}}x_{j}],x_{2},\cdots,\acute{x}_{j},\cdots,x_{p+1}))\\&\quad-\phi(\sum_{i=2}^{p+1}(-1)^{i+1}[x_{i\lambda_{i}}f_{\lambda_{2},\cdots,\lambda_{i},\cdots,\lambda_{p+1}}(Px_{1},x_{2},\cdots,\acute{x}_{i},\cdots,x_{p+1})]\\&\quad+\sum_{2\leq i<j\leq p+1}(-1)^{i+j}f_{\lambda_{i}+\lambda_{j},\lambda_{1},\cdots,\lambda_{i},\cdots,\lambda_{j},\cdots,\lambda_{p}}([x_{i\lambda_{i}}x_{j}],Px_{1},x_{2},\cdots,\acute{x}_{i},\cdots,\acute{x}_{j},\cdots,x_{p+1})).\end{aligned}\] 

Next

 \[\begin{aligned}\partial_{AO}^{p}(\xi^{p}(f))_{\lambda_{1},\cdots,\lambda_{p}}&(x_{1},x_{2},x_{3},\cdots,x_{p+1})\\&=\sum_{i=1}^{p+1}(-1)^{i+1}\rho(P(x_{i}))_{\lambda_{i}}(\xi^{p}f)_{\lambda_{1},\cdots,\lambda_{i},\cdots,\lambda_{p}}(x_{1},\cdots,\acute{x}_{i},\cdots,x_{p+1})\\&\quad+\sum_{1\leq i<j\leq p+1}(-1)^{i+j}(\xi^{p}f)_{\lambda_{i}+\lambda_{j},\lambda_{1},\cdots,\lambda_{i},\cdots,\lambda_{j},\cdots,\lambda_{p}}([P(x_{i})_{\lambda_{i}}x_{j}],x_{1},\cdots,\acute{x}_{i},\cdots,\acute{x}_{j},\cdots,x_{p+1})\\&=\sum_{i=1}^{p+1}(-1)^{i+1}\rho(P(x_{i}))_{\lambda_{i}}f_{\lambda_{1},\lambda_{2},\cdots,\lambda_{i},\cdots,\lambda_{p}}(Px_{1},Px_{2},\cdots,P\acute{x}_{i},\cdots,Px_{p+1})\\&\quad-\sum_{i=1}^{p+1}(-1)^{i+1}\rho(P(x_{i}))_{\lambda_{i}}(\phi\circ f)_{\lambda_{1},\lambda_{2},\cdots,\lambda_{i},\cdots,\lambda_{p}}(Px_{1},x_{2},\cdots,\acute{x}_{i},\cdots,x_{p+1})\\&\quad+\sum_{1\leq i<j\leq p+1}(-1)^{i+j}f_{\lambda_{i}+\lambda_{j},\lambda_{1},\cdots,\lambda_{i},\cdots,\lambda_{j},\cdots,\lambda_{p}}([P(x_{i})_{\lambda_{i}}x_{j}],Px_{1},Px_{2},\cdots P\acute{x}_{i},\cdots P\acute{x}_{j},\cdots,Px_{p})\\&\quad-\sum_{1\leq i<j\leq p+1}(-1)^{i+j}(\phi\circ f)_{\lambda_{i}+\lambda_{j},\lambda_{1},\lambda_{2},\cdots,\lambda_{i},\cdots,\lambda_{p}}([P(x_{i})_{\lambda_{i}}x_{j}],x_{1},x_{2},\cdots\acute{x}_{i},\cdots\acute{x}_{j},\cdots,x_{p}).\end{aligned}\] 

By using Eq. \( (7) \) , we see that

 \[\xi^{p+1}(\delta^{p}(f))_{\lambda_{1},\lambda_{2},\cdots,\lambda_{p}}(x_{1},x_{2},\cdots,x_{p+1})=\partial^{p}(\xi^{p}(f))_{\lambda_{1},\lambda_{2},\cdots,\lambda_{p}}(x_{1},x_{2},\cdots,x_{p+1}).\] 

By now we complete the proof.
