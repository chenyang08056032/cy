
 \[\begin{align*}&\omega(a_{2}a_{3},a_{1})+\omega(a_{1}a_{3},a_{2})-\omega(a_{3}a_{1},a_{2})-\omega(a_{2}a_{1},a_{3})\\=&\langle r^{\sharp}(\xi_{2})r^{\sharp}(\xi_{3}),\xi_{1}\rangle+\langle r^{\sharp}(\xi_{1})r^{\sharp}(\xi_{3}),\xi_{2}\rangle-\langle r^{\sharp}(\xi_{3})r^{\sharp}(\xi_{1}),\xi_{2}\rangle-\langle r^{\sharp}(\xi_{2})r^{\sharp}(\xi_{1}),\xi_{3}\rangle\\=&\sum_{i,j}\left(\langle\xi_{1},x_{i}x_{j}\rangle\langle\xi_{2},y_{i}\rangle\langle\xi_{3},y_{j}\rangle+\langle\xi_{1},x_{i}\rangle\langle\xi_{2},y_{i}x_{j}\rangle\langle\xi_{3},y_{j}\rangle\right.\\&\left.\quad-\langle\xi_{1},x_{j}\rangle\langle\xi_{2},x_{i}y_{j}\rangle\langle\xi_{3},y_{i}\rangle\right)-\langle\xi_{1},x_{j}\rangle\langle\xi_{2},x_{i}\rangle\langle\xi_{3},y_{i}y_{j}\rangle\right)\\=&\langle\xi_{1}\otimes\xi_{2}\otimes\xi_{3},r_{12}r_{13}+r_{12}r_{23}-r_{23}r_{12}-r_{23}r_{13}\rangle.\end{align*}\] 

That is, \(\omega(a_{2}a_{3},a_{1})+\omega(a_{1}a_{3},a_{2})-\omega(a_{3}a_{1},a_{2})-\omega(a_{2}a_{1},a_{3})=0\) for any \(a_{1},a_{2},a_{3}\in A\) if and only if \(\langle\xi_{1}\otimes\xi_{2}\otimes\xi_{3},\|r,r\|\rangle=0\) for any \(\xi_{1},\xi_{2},\xi_{3}\in A^{*}\), if and only if \(\|r,r\|=0\). The proof is finished. \(\square\)

Next, we show that each relative Rota-Baxter operator induce an anti-Leibniz bialgebra.

Proposition 3.20. Let  \( (A,\cdot) \)  be an anti-Leibniz algebra and  \( (M,\mathfrak{l},\mathfrak{r}) \)  be a bimodule over  \( (A,\cdot) \) . Let  \( P:M\to A \)  be a linear map which is identified as an element in  \( (A\ltimes M^{*})\otimes(A\ltimes M^{*}) \)  through  \( \operatorname{Hom}(M,A)\cong A\otimes M^{*}\subset(A\ltimes M^{*})\otimes(A\ltimes M^{*}) \) . Then  \( r:=P+\tau(P) \)  is a symmetric solution of the aLYBE in semi-direct product anti-Leibniz algebra  \( A\ltimes M^{*} \)  if and only if P is a relative Rota-Baxter operator of  \( (A,\cdot) \)  associated to  \( (M,\mathfrak{l},\mathfrak{r}) \) .

Proof. Let \(\{e_{1},e_{2},\cdots,e_{n}\}\) be a basis of \(M\) and \(\{f_{1},f_{2},\cdots,f_{n}\}\) be its dual basis in \(M^{*}\). Then \(P=\sum_{i=1}^{n}P(e_{i})\otimes f_{i}\in(A\ltimes M^{*})\otimes(A\ltimes M^{*})\), \(r=P-\tau(P)=\sum_{i=1}^{n}(P(e_{i})\otimes f_{i}+f_{i}\otimes P(e_{i}))\), and

 \[\begin{align*}r_{12}r_{13}&=\sum_{i,j=1}^{n}P(e_{i})P(e_{j})\otimes f_{i}\otimes f_{j}-\mathfrak{l}^{*}(P(e_{i}))(f_{j})\otimes f_{i}\otimes P(e_{j})-(\mathfrak{l}^{*}-\mathfrak{r}^{*})(P(e_{j}))(f_{i})\otimes P(e_{i})\otimes f_{j},\\r_{12}r_{23}&=\sum_{i,j=1}^{n}P(e_{i})\otimes(\mathfrak{l}^{*}-\mathfrak{r}^{*})(P(e_{j}))(f_{i})\otimes f_{j}-f_{i}\otimes P(e_{i})P(e_{j})\otimes f_{j}+f_{i}\otimes\mathfrak{l}^{*}(P(e_{i}))(f_{j})\otimes P(e_{j}),\\r_{23}r_{12}&=\sum_{i,j=1}^{n}P(e_{j})\otimes\mathfrak{l}^{*}(P(e_{i}))(f_{j})\otimes f_{i}-f_{j}\otimes P(e_{i})P(e_{j})\otimes f_{i}+f_{j}\otimes(\mathfrak{l}^{*}-\mathfrak{r}^{*})(P(e_{j}))(f_{i})\otimes P(e_{i}),\\r_{23}r_{13}&=\sum_{i,j=1}^{n}-f_{j}\otimes P(e_{i})\otimes(\mathfrak{l}^{*}-\mathfrak{r}^{*})(P(e_{j}))(f_{i})-P(e_{j})\otimes f_{i}\otimes\mathfrak{l}^{*}(P(e_{i}))(f_{j})+f_{i}\otimes f_{j}\otimes P(e_{i})P(e_{j}).\end{align*}\] 

Note that  \( \mathfrak{l}^{*}(P(e_{i}))(f_{j})=\sum_{k=1}f_{j}(\mathfrak{l}(P(e_{i}))(e_{k}))f_{k} \) , we get

 \[\begin{align*}\sum_{i,j=1}^{n}P(e_{j})\otimes\mathfrak{l}^{*}(P(e_{i}))(f_{j})\otimes f_{i}=\sum_{i,j,k=1}^{n}f_{j}(\mathfrak{l}(P(e_{i}))(e_{k}))P(e_{j})\otimes f_{k}\otimes f_{i}\\=\sum_{i,j=1}^{n}P\Big(\sum_{k=1}^{n}f_{k}(\mathfrak{l}(P(e_{i}))(e_{j}))e_{k}\Big)\otimes f_{j}\otimes f_{i}=\sum_{i,j=1}^{n}P(\mathfrak{l}(P(e_{i}))(e_{j}))\otimes f_{j}\otimes f_{i}.\end{align*}\] 

Similarly, we also have  \( \sum_{i,j=1}^{n}f_{j}\otimes\mathfrak{l}^{*}(P(e_{j}))(f_{i})\otimes P(e_{i})=f_{j}\otimes f_{i}\otimes P(\mathfrak{l}(P(e_{i}))(e_{j})),\sum_{i,j=1}^{n}P(e_{i})\otimes\mathfrak{r}^{*}(P(e_{j}))(f_{i})\otimes f_{j}=\sum_{i,j=1}^{n}P(\mathfrak{r}(P(e_{j}))(e_{i}))\otimes f_{i}\otimes f_{j} \)  and  \( \sum_{i,j=1}^{n}f_{i}\otimes\mathfrak{r}^{*}(P(e_{i}))(f_{j})\otimes P(e_{i})=f_{i}\otimes f_{j}\otimes P(\mathfrak{r}(P(e_{j}))(e_{i})) \) . Thus

 \[\begin{align*}r_{12}r_{13}+r_{12}r_{23}-r_{23}r_{12}-r_{23}r_{13}=\sum_{i,j=1}^{n}\Big((P(e_{i})P(e_{j})-P(\mathfrak{r}(P(e_{j}))(e_{i})-P(\mathfrak{l}(P(e_{i}))(e_{j})))\otimes f_{i}\otimes f_{j}\\ +f_{i}\otimes f_{j}\otimes(P(\mathfrak{l}(P(e_{i}))(e_{j}))+P(\mathfrak{r}(P(e_{j}))(e_{i}))-P(e_{i})P(e_{j}))\Big).\end{align*}\] 

Thus, r is a solution of the aLYBE in  \( A \ltimes M^{*} \)  if and only if P is a relative Rota-Baxter operator of  \( (A, \cdot) \)  associated to  \( (M, \mathfrak{l}, \mathfrak{r}) \) .

As a direct conclusion, we have

Corollary 3.21. Let  \( (A,\cdot) \)  be an anti-Leibniz algebra and  \( (M,\mathfrak{l},\mathfrak{r}) \)  be a bimodule over  \( (A,\cdot) \) . If  \( P:M\to A \)  is a relative Rota-Baxter operator of  \( (A,\cdot) \)  associated to  \( (M,\mathfrak{l},\mathfrak{r}) \) , then there is an anti-Leibniz algebra  \( (A\ltimes M^{*},\Delta_{r}) \) , where  \( \Delta_{r} \)  is given by Eq. (3.1) for  \( r=P+\tau(P) \) .
