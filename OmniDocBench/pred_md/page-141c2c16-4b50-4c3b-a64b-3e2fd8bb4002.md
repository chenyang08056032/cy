function Z = phasel_afterexzfft(xx, fi, Nw, Lw, fs1, nx1, nx2)
xx = xx(:,).';
ddf = fs1 / Nw;
u1 = xx(1 : Nw);
u2 = xx(Lw + 1 : Lw + Nw);
w = hanning(Nw);
U1 = fft(u1 . * w');
U1 = fftshift(U1);
U2 = fft(u2 . * w');
U2 = fftshift(U2);
if nx1 - fi <= 0, nx1 = fi; end
n1 = floor((nx1 - fi) / ddf) + 1;
n2 = floor((nx2 - fi) / ddf) + 1;
