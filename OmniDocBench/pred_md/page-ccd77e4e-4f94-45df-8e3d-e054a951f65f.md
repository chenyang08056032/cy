
For some  \(  x_{i} \notin ((\mu^{-1}(\hat{Y}\tau^{-\varepsilon}))^{-1}, \mu^{-1}(\hat{Y}\tau^{-\varepsilon}))  \) , we may as well assume that there exist  \( i_{1} \in \{m_{1}, \cdots, m_{s_{1}}\} \)  and  \( i_{2} \in \{n_{1}, \cdots, n_{s_{2}}\} \)  such that  \(  x_{i_{1}} > \mu^{-1}(\hat{Y}\tau^{-\varepsilon})  \)  and  \(  x_{i_{2}} < (\mu^{-1}(\hat{Y}\tau^{-\varepsilon}))^{-1}  \) , respectively. Let  \( s_{1} := \{m_{1}, \cdots, m_{s_{1}}\} \)  and  \( s_{2} := \{n_{1}, \cdots, n_{s_{2}}\} \) . Then

 \[\langle D f_{\tau}(x)y,y\rangle+m_{0}|D g_{\tau}(x)y|^{2}=\sum_{i=1,i\notin s_{1}\cup s_{2}}^{d}\left(\frac{\partial f(x)}{\partial x_{i}}y_{i}^{2}+m_{0}\Bigg|\sum_{j=1}^{m}\frac{\partial g^{j}(x)}{\partial x_{i}}y_{i}\Bigg|^{2}\right)+M_{1}+M_{2},\] 

where

 \[M_{1}=\sum_{i=m_{1}}^{m_{s_{1}}}\left(\frac{\partial f_{\tau}(x)}{\partial x_{i}}y_{i}^{2}+m_{0}\Big|\sum_{j=1}^{m}\frac{\partial g_{\tau}^{j}(x)}{\partial x_{i}}y_{i}\Big|^{2}\right)\] 

and

 \[M_{2}=\sum_{i=m_{1}}^{n_{s_{2}}}\left(\frac{\partial f_{\tau}(x)}{\partial x_{i}}y_{i}^{2}+m_{0}\Big|\sum_{j=1}^{m}\frac{\partial g_{\tau}^{j}(x)}{\partial x_{i}}y_{i}\Big|^{2}\right).\] 

We obtain the following estimation of  \( M_{1} \)  from (D1) that

 \[\begin{align*}M_{1}&=\sum_{i=m_{1}}^{m_{s_{1}}}\left(\frac{\partial f(\Phi(x_{1}),\cdots,\Phi(x_{d}))}{\partial\Phi(x_{i})}y_{i}^{2}\cdot\frac{1}{1+(x_{i}-\mu^{-1}(\hat{Y}\tau^{-\varepsilon}))^{\bar{v}}}\right.\\&\quad\left.+m_{0}\Big|\sum_{j=1}^{m}\frac{\partial g^{j}(\Phi(x_{1}),\cdots,\Phi(x_{d}))}{\partial\Phi(x_{i})}y_{i}\cdot\frac{1}{1+(x_{i}-\mu^{-1}(\hat{Y}\tau^{-\varepsilon}))^{\bar{v}}}\Big|^{2}\right)\\&=\sum_{i=m_{1}}^{m_{s_{1}}}\left(\frac{1}{(1+(x_{i}-\mu^{-1}(\hat{Y}\tau^{-\varepsilon}))^{\bar{v}})^{2}}\cdot\left[\frac{\partial f(\Phi(x_{1}),\cdots,\Phi(x_{d}))}{\partial\Phi(x_{i})}y_{i}^{2}\right.\right.\\&\quad\left.\left.+m_{0}\Big|\sum_{j=1}^{m}\frac{\partial g^{j}(\Phi(x_{1}),\cdots,\Phi(x_{d}))}{\partial\Phi(x_{i})}y_{i}\Big|^{2}\right.\right.\\&\quad\left.\left.\leq\sum_{i=m_{1}}^{m_{s_{1}}}\left[\frac{\partial f(\Phi(x_{1}),\cdots,\Phi(x_{d}))}{\partial\Phi(x_{i})}y_{i}^{2}+m_{0}\Big|\sum_{j=1}^{m}\frac{\partial g_{j}(\Phi(x_{1}),\cdots,\Phi(x_{d}))}{\partial\Phi(x_{i})}y_{i}\Big|^{2}\right]\right.\\&\quad\left.\leq C\sum_{i=m_{1}}^{m_{s_{1}}}y_{i}^{2}.\right.\end{align*}\] 

For  \( M_{2} \) , we analogously obtain

 \[M_{2}\leq\sum_{i=m_{1}}^{n_{s_{2}}}\left[\frac{\partial f(\Phi(x_{1}),\cdots,\Phi(x_{d}))}{\partial\Phi(x_{i})}y_{i}^{2}+m_{0}\Big|\sum_{j=1}^{m}\frac{\partial g^{j}(\Phi(x_{1}),\cdots,\Phi(x_{d}))}{\partial\Phi(x_{i})}y_{i}\Big|^{2}\right]\leq C\sum_{i=n_{1}}^{n_{s_{2}}}y_{i}^{2}.\] 

Substituting (D3) and (D4) into (D2) yields that for sufficiently large  \( m_{0} > 0 \)  and any  \( x, y \in R^{d} \) ,

 \[\langle D f_{\tau}(x)y,y\rangle+m_{0}|D g_{\tau}(x)y|^{2}\leq C|y|^{2}.\]
