Next, we derive the model likelihood  \( \log \mu^{\theta} \) . Notice that the marginal density  \( p_{t}^{\theta}(x) \)  obeys the Fokker Plank equation

 \[\begin{align*}&\frac{\partial p_{t}^{\theta}(x)}{\partial t}=-\nabla\cdot\left(p_{t}^{\theta}(x)\sigma Z_{t}^{\theta}(x)\right)+\frac{\sigma^{2}}{2}\Delta p_{t}^{\theta}(x)\\\Rightarrow\frac{\partial\log p_{t}^{\theta}}{\partial t}&=\frac{1}{p_{t}^{\theta}}\Big(-p_{t}^{\theta}\nabla\cdot\left(\sigma Z_{t}^{\theta}\right)-\nabla p_{t}^{\theta}\cdot\sigma Z_{t}^{\theta}+\frac{\sigma^{2}}{2}\Delta p_{t}^{\theta}\Big)\\&=-\nabla\cdot\left(\sigma Z_{t}^{\theta}\right)-\nabla\log p_{t}^{\theta}\cdot\sigma Z_{t}^{\theta}+\frac{\sigma^{2}}{2}\Delta\log p_{t}^{\theta}+\frac{1}{2}\|\sigma\nabla\log p_{t}^{\theta}\|^{2},\end{align*}\] 

where the last equality follows by Equation 15. Now, invoke Itô formula

 \[\begin{align*}\mathrm{d}\log p_{t}^{\theta}&=\frac{\partial\log p_{t}^{\theta}}{\partial t}\mathrm{d}t+\left[\nabla\log p_{t}^{\theta}\cdot\sigma Z_{t}^{\theta}+\frac{\sigma^{2}}{2}\Delta\log p_{t}^{\theta}\right]\mathrm{d}t+\sigma\nabla\log p_{t}^{\theta}\cdot\mathrm{d}W_{t}\\&=\left[-\nabla\cdot\left(\sigma Z_{t}^{\theta}\right)+\sigma^{2}\Delta\log p_{t}^{\theta}+\frac{1}{2}\|\sigma\nabla\log p_{t}^{\theta}\|^{2}\right]\mathrm{d}t+\sigma\nabla\log p_{t}^{\theta}\cdot\mathrm{d}W_{t},\end{align*}\] 

where the last equality follows by Equation 31. Equation 32 facilitates a computation of the parametrized model likelihood

 \[\begin{align*}\log\mu^{\theta}(X_{0})&=\mathbb{E}_{p_{1}^{\theta}}[\log p_{1}^{\theta}(X_{1}^{\theta})]-\int_{0}^{1}\mathbb{E}_{p_{t}^{\theta}}\left[\mathrm{d}\log p_{t}^{\theta}\right]\\&=\mathbb{E}_{p_{1}^{\theta}}[\log p_{1}^{\theta}(X_{1}^{\theta})]-\int_{0}^{1}\mathbb{E}_{p_{t}^{\theta}}\left[-\nabla\cdot\left(\sigma Z_{t}^{\theta}\right)+\sigma^{2}\Delta\log p_{t}^{\theta}+\frac{1}{2}\|\sigma\nabla\log p_{t}^{\theta}\|^{2}\right]\mathrm{d}t\\&=\mathbb{E}_{p_{1}^{\theta}}[\log p_{1}^{\theta}(X_{1}^{\theta})]-\int_{0}^{1}\mathbb{E}_{p_{t}^{\theta}}\left[-\nabla\cdot\left(\sigma Z_{t}^{\theta}\right)-\frac{1}{2}\|\sigma\nabla\log p_{t}^{\theta}\|^{2}\right]\mathrm{d}t,\end{align*}\] 

where the last equality follows by Equation 16. Although Equation 33 implies that the parametrized model likelihood is intractable, due to the unknown score function  \( \nabla \log p_{t}^{\theta} \) , combining this likelihood with the KL divergence in Equation 29 yields a tractable objective:

 \[\begin{aligned}&D_{\mathrm{KL}}(p^{\theta}||p^{\phi})-\log\mu^{\theta}(X_{0})\\&=D_{\mathrm{KL}}(p_{1}^{\theta}||\nu)+\int_{0}^{1}\mathbb{E}_{p_{t}^{\theta}}\left[\frac{1}{2}||Z_{t}^{\theta}-\sigma\nabla\log p_{t}^{\theta}+\widehat{Z}_{t}^{\phi}||^{2}\right]\mathrm{d}t\\&\quad-\mathbb{E}_{p_{1}^{\theta}}[\log p_{1}^{\theta}(X_{1}^{\theta})]+\int_{0}^{1}\mathbb{E}_{p_{t}^{\theta}}\left[-\nabla\cdot\left(\sigma Z_{t}^{\theta}\right)-\frac{1}{2}||\sigma\nabla\log p_{t}^{\theta}||^{2}\right]\mathrm{d}t\\&=-\mathbb{E}_{p_{1}^{\theta}}[\log\nu(X_{1}^{\theta})]+\int_{0}^{1}\mathbb{E}_{p_{t}^{\theta}}\left[\frac{1}{2}||Z_{t}^{\theta}-\sigma\nabla\log p_{t}^{\theta}+\widehat{Z}_{t}^{\phi}||^{2}-\nabla\cdot\left(\sigma Z_{t}^{\theta}\right)-\frac{1}{2}||\sigma\nabla\log p_{t}^{\theta}||^{2}\right]\mathrm{d}t\\&=-\mathbb{E}_{p_{1}^{\theta}}[\log\nu(X_{1}^{\theta})]+\int_{0}^{1}\mathbb{E}_{p_{t}^{\theta}}\left[\frac{1}{2}||Z_{t}^{\theta}+\widehat{Z}_{t}^{\phi}||^{2}-\nabla\cdot\left(\sigma Z_{t}^{\theta}\right)-\left(Z_{t}^{\theta}+\widehat{Z}_{t}^{\phi}\right)\cdot\sigma\nabla\log p_{t}^{\theta}\right]\mathrm{d}t\\&=-\mathbb{E}_{p_{1}^{\theta}}[\log\nu(X_{1}^{\theta})]+\int_{0}^{1}\mathbb{E}_{p_{t}^{\theta}}\left[\frac{1}{2}||Z_{t}^{\theta}+\widehat{Z}_{t}^{\phi}||^{2}-\nabla\cdot\left(\sigma Z_{t}^{\theta}\right)+\nabla\cdot\left(\sigma Z_{t}^{\theta}+\sigma\widehat{Z}_{t}^{\phi}\right)\right]\mathrm{d}t\\&=-\mathbb{E}_{p_{1}^{\theta}}[\log\nu(X_{1}^{\theta})]+\int_{0}^{1}\mathbb{E}_{p_{t}^{\theta}}\left[\frac{1}{2}||Z_{t}^{\theta}+\widehat{Z}_{t}^{\phi}||^{2}+\nabla\cdot\left(\sigma\widehat{Z}_{t}^{\phi}\right)\right]\mathrm{d}t\end{aligned}\]
